// App.js import React, { useState } from 'react'; import { SafeAreaView, Text, TextInput, TouchableOpacity, View, StyleSheet, ScrollView } from 'react-native'; import * as Speech from 'expo-speech'; import * as Voice from '@react-native-voice/voice';

export default function App() { const [word, setWord] = useState(''); const [syllables, setSyllables] = useState([]); const [listening, setListening] = useState(false);

const splitIntoSyllables = (text) => { // basic syllable split - improve with API or library later return text .toLowerCase() .replace(/[^a-z]/g, '') .split(/(?=[aeiouy])/); };

const handleSoundOut = () => { if (!word.trim()) return; const parts = splitIntoSyllables(word); setSyllables(parts); Speech.speak(word, { rate: 0.7 }); };

const handlePlaySyllables = async () => { for (let part of syllables) { await new Promise((res) => { Speech.speak(part, { rate: 0.6, onDone: res }); }); } };

const startListening = async () => { setListening(true); try { Voice.onSpeechResults = (e) => { const said = e.value && e.value[0]; if (said) setWord(said); setListening(false); }; await Voice.start('en-US'); } catch (e) { console.log(e); setListening(false); } };

return ( <SafeAreaView style={styles.container}> <Text style={styles.header}>🔊 Sound It Out</Text> <TextInput
style={styles.input}
placeholder="Type a word..."
value={word}
onChangeText={setWord}
/> <View style={styles.buttonRow}> <TouchableOpacity style={styles.button} onPress={handleSoundOut}> <Text style={styles.buttonText}>Sound It Out</Text> </TouchableOpacity> <TouchableOpacity style={styles.button} onPress={handlePlaySyllables}> <Text style={styles.buttonText}>Play Syllables</Text> </TouchableOpacity> </View> <TouchableOpacity style={styles.micButton} onPress={startListening}> <Text style={styles.micText}>{listening ? '🎤 Listening...' : '🎤 Speak Word'}</Text> </TouchableOpacity> <ScrollView style={styles.resultContainer}> {syllables.length > 0 && ( <Text style={styles.syllables}>{syllables.join(' • ')}</Text> )} </ScrollView> </SafeAreaView> ); }

const styles = StyleSheet.create({ container: { flex: 1, padding: 20, backgroundColor: '#fdfdfd' }, header: { fontSize: 28, fontWeight: 'bold', textAlign: 'center', marginVertical: 20 }, input: { borderWidth: 1, borderColor: '#aaa', borderRadius: 12, padding: 12, fontSize: 18, marginBottom: 20 }, buttonRow: { flexDirection: 'row', justifyContent: 'space-between', marginBottom: 20 }, button: { backgroundColor: '#4a90e2', padding: 12, borderRadius: 12, flex: 1, marginHorizontal: 4 }, buttonText: { color: '#fff', fontSize: 16, textAlign: 'center' }, micButton: { backgroundColor: '#f06d6d', padding: 14, borderRadius: 12, marginBottom: 20 }, micText: { color: '#fff', fontSize: 18, textAlign: 'center' }, resultContainer: { flex: 1, marginTop: 10 }, syllables: { fontSize: 24, fontWeight: '600', color: '#333', textAlign: 'center' } });
